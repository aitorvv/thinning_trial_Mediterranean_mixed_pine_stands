Summary

This document generates and exports publication-ready high-resolution figures for the Ampudia Mediterranean mixed forest thinning trial data paper:

  1. Stand-level metrics summary (Density, Basal Area, Dominant Height, and Volume across plots and inventories).
  2. Tree spatial distribution and status (Color: Status, Shape: Species).
  3. Tree spatial distribution and species (Color: Species, Shape: Status).
  4. Multi-scale geographical location map (European context, Spanish context, and Stand plot distribution).

1. Stand Level Metrics by Plot and Inventory

# Load dataset
plots_desc_path <- ifelse(file.exists("../plots_description.csv"), "../plots_description.csv", "final/plots_description.csv")
plots_desc <- read.csv(plots_desc_path)

# Create simplified plot ID / label (0, 15, 30, 45 % thinning)
plots_desc$plot_num <- factor(plots_desc$plot_id - 2434000, levels = 1:4, labels = c("control", "15", "30", "45"))

# Prepare long-format data for faceted plotting
plots_long <- plots_desc %>%
  select(inventory_id, plot_num, n, g, ho, sfni_vcc) %>%
  rename(
    `N (trees/ha)` = n,
    `G (m²/ha)` = g,
    `Ho (m)` = ho,
    `VCC (m³/ha)` = sfni_vcc
  ) %>%
  pivot_longer(
    cols = c(`N (trees/ha)`, `G (m²/ha)`, `Ho (m)`, `VCC (m³/ha)`), 
    names_to = "variable", 
    values_to = "value"
  ) %>%
  mutate(variable = factor(variable, levels = c("N (trees/ha)", "G (m²/ha)", "Ho (m)", "VCC (m³/ha)")))

# Create joint graph
joint_plot <- ggplot(plots_long, aes(x = plot_num, y = value, fill = as.factor(inventory_id))) +
  geom_col(position = position_dodge(width = 0.8), width = 0.7) +
  facet_wrap(~ variable, scales = "free_y") +
  labs(
    # title = "Stand level metrics per plot and inventory",
    x = "Thinning intensity (%)",
    y = NULL,
    fill = "Inventory"
  ) +
  scale_fill_brewer(palette = "Dark2") +
  theme_minimal() +
  theme(
    # plot.title = element_text(face = "bold", size = 14, hjust = 0.5),
    strip.text = element_text(face = "bold", size = 11),
    legend.position = "bottom"
  )

print(joint_plot)
# Save high-resolution publication figure
ggsave(filename = "figure_1_stand_level_metrics.png", plot = joint_plot, dpi = 300, width = 8, height = 6)

# Clean memory
rm(plots_desc, plots_long, plots_desc_path)

2. Tree Spatial Distribution (Color: Status, Shape: Species)

# Load tree inventory
trees_inv_path <- ifelse(file.exists("../trees_inventory.csv"), "../trees_inventory.csv", "final/trees_inventory.csv")
trees_inv <- read.csv(trees_inv_path)

# Prepare spatial data for all inventories using tree_status
trees_spatial_all <- trees_inv %>%
  filter(!is.na(x) & !is.na(y)) %>%
  mutate(
    plot_num = factor(plot_id - 2434000, levels = 1:4, labels = c("control", "15", "30", "45")),
    Status = case_when(
      tree_status == "I" ~ "Ingrowth",
      tree_status == "C" ~ "Harvested",
      tree_status == "M" ~ "Dead",
      TRUE ~ "Alive"
    ),
    Status = factor(Status, levels = c("Alive", "Ingrowth", "Harvested", "Dead"))
  )

# Reusable plot function (Version 1: Color = Status, Shape = Species)
create_inv_plot_v1 <- function(inv_id, title) {
  df <- trees_spatial_all %>% filter(inventory_id == inv_id)
  
  ggplot(df, aes(x = x, y = y, color = Status, shape = species_name)) +
    geom_point(size = 2.5, alpha = 0.8) +
    scale_color_manual(
      values = c(
        "Alive" = "#4daf4a",      # Green
        "Ingrowth" = "#377eb8",   # Blue
        "Harvested" = "#ff7f00",  # Orange
        "Dead" = "#e41a1c"        # Red
      ), 
      drop = FALSE
    ) +
    scale_shape_manual(
      breaks = c("Pinus halepensis", "Pinus pinea", "Cupressus sempervirens", "Quercus faginea", "Quercus ilex"),
      values = c(
        "Pinus halepensis" = 16,
        "Pinus pinea" = 17,
        "Cupressus sempervirens" = 18,
        "Quercus faginea" = 15,
        "Quercus ilex" = 3
      ),
      labels = c(
        expression(italic("P. halepensis")),
        expression(italic("P. pinea")),
        expression(italic("C. sempervirens")),
        expression(italic("Q. faginea")),
        expression(italic("Q. ilex"))
      ),
      drop = FALSE
    ) +
    guides(
      color = guide_legend(override.aes = list(shape = 16, size = 3)),
      shape = guide_legend(override.aes = list(color = "black", size = 3))
    ) +
    labs(
      title = title,
      color = "Status: ",
      shape = "Species: "
    ) +
    theme_minimal() +
    theme(
      plot.title = element_text(face = "bold", size = 14, hjust = 0.5),
      axis.title = element_blank(),
      axis.text = element_blank(),
      axis.ticks = element_blank(),
      panel.grid = element_blank(),
      panel.border = element_rect(color = "gray80", fill = NA),
      legend.position = "bottom",
      legend.box = "horizontal",
      legend.text = element_text(size = 9),
      legend.title = element_text(face = "bold", size = 9.5)
    )
}

# Create individual plots
p1_v1 <- create_inv_plot_v1(1, "Inventory 1 (2008)")
p2_v1 <- create_inv_plot_v1(2, "Inventory 2 (2014)")
p3_v1 <- create_inv_plot_v1(3, "Inventory 3 (2023)")

# Extract complete legend from p2 (contains all statuses)
legend_grob_v1 <- get_legend(p2_v1 + theme(legend.position = "bottom", legend.box = "horizontal"))

# Combine plots with legend
plots_v1 <- ggarrange(p1_v1, p2_v1, p3_v1, nrow = 3, ncol = 1, legend = "none")
combined_map_v1 <- ggarrange(plots_v1, legend_grob_v1, nrow = 2, heights = c(1, 0.1))

print(combined_map_v1)
# Save high-resolution publication figure
ggsave(filename = "figure_2_tree_spatial_distribution_status.png", plot = combined_map_v1, dpi = 300, width = 10, height = 10)

# Clean memory
rm(p1_v1, p2_v1, p3_v1, legend_grob_v1, plots_v1, create_inv_plot_v1, trees_inv_path)
